"""Reflow one paragraph of a text file in place (AMEND stage; FINALISE's reflow.py plus a continuation indent).

Usage: reflow2.py FILE FIRST_LINE_PREFIX LAST_LINE_SUFFIX [WIDTH] [CONT_INDENT]

The paragraph starts at the one line that starts with FIRST_LINE_PREFIX (after its indentation) and ends at the
first line at or after it that ends with LAST_LINE_SUFFIX. Its words are rewrapped at WIDTH (default 120); the first
line keeps its indentation, later lines get CONT_INDENT spaces (default: the first line's indentation, plus 2 when
the first line is a "- " bullet). Whitespace only: the words and their order are unchanged (checked). The file must
be ASCII; the script refuses if either line is not found as required.
"""

import sys
import textwrap


def main() -> int:
    path, first, last = sys.argv[1], sys.argv[2], sys.argv[3]
    width = int(sys.argv[4]) if len(sys.argv) > 4 else 120
    with open(path, encoding="ascii") as handle:
        lines = handle.read().split("\n")
    starts = [i for i, line in enumerate(lines) if line.lstrip().startswith(first)]
    if len(starts) != 1:
        print(f"refusing: {len(starts)} lines start with {first!r}")
        return 2
    start = starts[0]
    ends = [i for i in range(start, len(lines)) if lines[i].rstrip().endswith(last)]
    if not ends:
        print(f"refusing: no line ends with {last!r} after line {start + 1}")
        return 2
    end = ends[0]
    indent = lines[start][: len(lines[start]) - len(lines[start].lstrip())]
    default_cont = len(indent) + (2 if lines[start].lstrip().startswith("- ") else 0)
    cont = " " * (int(sys.argv[5]) if len(sys.argv) > 5 else default_cont)
    words = " ".join(line.strip() for line in lines[start : end + 1])
    wrapped = textwrap.wrap(words, width=width, initial_indent=indent, subsequent_indent=cont,
                            break_on_hyphens=False, break_long_words=False)
    if " ".join(w.strip() for w in wrapped).split() != words.split():
        print("refusing: the words changed")
        return 2
    lines[start : end + 1] = wrapped
    with open(path, "w", encoding="ascii") as handle:
        handle.write("\n".join(lines))
    print(f"reflowed lines {start + 1}-{end + 1} into {len(wrapped)} lines")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
