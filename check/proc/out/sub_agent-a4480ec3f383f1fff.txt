def _ls_slopes(xs, ys):
    import numpy as np

    xs = np.asarray(xs, dtype=np.float64)
    if xs.ndim == 1:
        xs = np.repeat(xs[:, None], ys.shape[1], axis=1)
    xc = xs - xs.mean(axis=0)
    return (xc * (ys - ys.mean(axis=0))).sum(axis=0) / (xc ** 2).sum(axis=0)
