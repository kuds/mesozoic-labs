   417	record_crash() {  # record_crash <name> <cause> <counts 0|1>
   418	  local name=$1 cause=$2 counts=$3 new=none reached=none early=0 n load
   419	  load=${S_OF[$name]}
   420	  if [[ -d "$RUNS/$name" ]]; then
   421	    reached=$(reached_step "$RUNS/$name")
   422	    load=$(load_step_of "$RUNS/$name" "${S_OF[$name]}")
   423	  fi
   424	  if (( counts == 1 )); then
   425	    if [[ "$reached" == none ]] || (( reached < load + EVAL_FREQ )); then early=1; counts=0; fi
   426	  fi
   427	  if [[ -d "$RUNS/$name" ]]; then
   428	    new=$(retire "$name" crashed)
   429	    RESUME_FROM[$name]=$RUNS/$new
   430	  fi
   431	  (( early )) && cause="EARLY CRASH, no window evaluation (last step $reached): $cause"
   432	  printf '%s\t%s\t%s\t%s\t%s\n' "$(date -u +%Y-%m-%dT%H:%M:%SZ)" "$name" "$new" "$cause" "$counts" >> "$CRASHES"
   433	  n=$(own_crashes "$name")
   434	  if (( early )); then
   435	    HALT=1
   436	    HALT_WHY+=("early crash of $name (runs/$new)")
   437	    note "$name: $cause; folder $new; not counted (own crashes $n of $MAX_CRASHES); no new start: the coordinator" \
   438	         "finds the cause from runs/$new/run_meta.json (error, traceback; never a process log) and fixes it"
   439	    return
   440	  fi
   441	  note "$name: CRASH ($cause; folder $new; own crash $n of $MAX_CRASHES$( ((counts == 0)) && echo ', not counted'))"
   442	  if (( n >= MAX_CRASHES )); then
   443	    if ! write_lost "$name" "$n own crashes, last: $cause"; then
   444	      note "lost.json could not be written for $name"
   445	      HALT=1
   446	      HALT_WHY+=("lost.json not written for $name")
   447	      return
   448	    fi
   449	    decide "$name" LOST "$n own crashes"
   450	  else
   451	    PENDING=("$name" "${PENDING[@]+"${PENDING[@]}"}")
   452	  fi
   453	}

   455	after_exit() {  # after_exit <name> <rc>
   456	  local name=$1 rc=$2 run=$RUNS/$1 crc reason new arm_ seed_ snap
   457	  if snap=$(exit_snapshot "$name"); then   # I1: before CHECK, the re-arm and any rename
   458	    note "$name: atimes at exit (I1, lstat): logs/atimes_at_exit/$name.attempt$(current_attempt "$name").tsv, $snap"
   459	  else
   460	    note "$name: the atime snapshot at exit failed (I1)"
   461	  fi
   462	  if (( rc == 0 || rc == POST_CHECK )); then
   463	    set +e
   464	    "${CHECK_CMD[@]}" "$run" >> "$LOGS/$name.check.log" 2>&1 9>&-
   465	    crc=$?
   466	    set -e
   467	    if (( crc == 0 && rc == 0 )); then
   468	      reason=$(meta_field "$run" stop_reason)
   469	      case "$reason" in
   470	        end) decide "$name" COUNTED "CHECK passed, stop reason end"; rearm "$name" ;;
   471	        collapse|hop_watch) decide "$name" STOPPED "CHECK passed, stop reason $reason"; rearm "$name" ;;
   472	        *) retire_void "$name" "CHECK passed but stop reason '$reason'" ;;
   473	      esac
   474	    elif (( crc == 4 )); then
   475	      retire_void "$name" "CHECK failed (logs/$name.check.log; run exit $rc)"
   476	    elif (( crc == 0 )); then
   477	      retire_void "$name" "run exit $rc (a post-run check failed) although CHECK passed (logs/$name.check.log)"
   478	    else
   479	      HALT=1
   480	      HALT_WHY+=("CHECK ERROR on $name")
