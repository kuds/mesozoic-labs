   179	    return datetime.now(timezone.utc).strftime("%Y-%m-%dT%H:%M:%SZ")

   182	def sha256_file(path) -> str:
   183	    h = hashlib.sha256()
   184	    with open(path, "rb") as fh:
   185	        for chunk in iter(lambda: fh.read(1 << 20), b""):
   186	            h.update(chunk)
   187	    return h.hexdigest()
   188	
   189	
   190	def sha256_bytes(b: bytes) -> str:
   191	    return hashlib.sha256(b).hexdigest()

   194	def require_under(path, root=MEASURE) -> Path:
   195	    """Every file this tool writes lies under LB/measure (the task's write rule)."""
   196	    p = Path(os.path.realpath(path))
   197	    r = Path(os.path.realpath(root))
   198	    if p != r and r not in p.parents:
   199	        raise Refusal(f"{path} is not under {root}: refused (this tool writes only under LB/measure)")
   200	    return p
