   242	def check_pins() -> list[str]:
   243	    """Every pinned file at its sha256, the measuring copy's seal values, the stack."""
   244	    out = []
   245	    for path, want in PINS.items():
   246	        try:
   247	            got = sha256_file(path)
   248	        except OSError as exc:
   249	            out.append(f"{path}: {exc}")
   250	            continue
   251	        if got != want:
   252	            out.append(f"{path}: sha256 {got} != pinned {want}")
   253	    try:
   254	        seal = json.loads(Path(CODE, ".r10m_seal.json").read_text())
   255	        if seal.get("commit") != CODE_COMMIT or seal.get("tree_digest") != CODE_TREE or seal.get("files") != CODE_FILES:
   256	            out.append(f"{CODE}: seal {seal.get('commit')} / {seal.get('tree_digest')} / {seal.get('files')} is not "
   257	                       f"{CODE_COMMIT} / {CODE_TREE} / {CODE_FILES}")
   258	    except (OSError, ValueError) as exc:
   259	        out.append(f"{CODE}: seal unreadable ({exc})")
   260	    try:
   261	        if json.loads(Path(ROUND_APPROVED).read_text()).get("branch_head") != CODE_COMMIT:
   262	            out.append(f"{ROUND_APPROVED}: branch_head is not {CODE_COMMIT}")
   263	    except (OSError, ValueError) as exc:
   264	        out.append(f"{ROUND_APPROVED}: unreadable ({exc})")
   265	    got = stack_versions()
   266	    out += [f"stack {k}: running {got.get(k)} != {v}" for k, v in STACK.items() if got.get(k) != v]
   267	    return out

   273	def lane_b():
   274	    """The round's lane B modules, loaded by path after their sha256 check: (r10round_common, ledger_r10round,
   275	    complete_r10round, score_r10round).  score_r10round loads the other three from its own folder (its _load)."""
   276	    if _LANE_B:
   277	        return _LANE_B["C"], _LANE_B["LG"], _LANE_B["CP"], _LANE_B["SC"]
   278	    names = ("r10round_common.py", "ledger_r10round.py", "complete_r10round.py", "score_r10round.py")
   279	    bad = [p for p in check_pins() if any(p.startswith(f"{LANE_B}/{n}") for n in names)]
   280	    if bad:
   281	        raise Refusal("lane B tools not at their pins: " + "; ".join(bad))
   282	    spec = importlib.util.spec_from_file_location("score_r10round", f"{LANE_B}/score_r10round.py")
   283	    mod = importlib.util.module_from_spec(spec)
   284	    sys.modules["score_r10round"] = mod
   285	    spec.loader.exec_module(mod)
   286	    for n in names:                        # the files read are still the pinned ones
   287	        if sha256_file(f"{LANE_B}/{n}") != PINS[f"{LANE_B}/{n}"]:
   288	            raise Refusal(f"{LANE_B}/{n} changed while it was loaded")
   289	    if Path(mod.C.__file__).resolve() != Path(LANE_B, "r10round_common.py").resolve():
   290	        raise Refusal(f"r10round_common was loaded from {mod.C.__file__}, not {LANE_B}")
   291	    _LANE_B.update(C=mod.C, LG=mod.LG, CP=mod.CP, SC=mod)
   292	    return mod.C, mod.LG, mod.CP, mod
