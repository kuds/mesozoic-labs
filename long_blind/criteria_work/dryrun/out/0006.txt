   252	def classify(name, arm, seed, check, csha, lost_entry):
   253	    """One entry's class from check.json and its lost.json entry (criteria RUN HANDLING)."""
   254	    out = dict(name=name, arm=arm, seed=seed, cls=UNDECIDED, T=None, cause=None, check_sha256=csha)
   255	    if lost_entry is not None:
   256	        cause = lost_entry.get("cause") if isinstance(lost_entry, dict) else None
   257	        detail = lost_entry.get("detail") if isinstance(lost_entry, dict) else None
   258	        out.update(cls=LOST, cause=f"lost.json: {cause or 'listed'}" + (f" ({detail})" if detail else ""))
   259	        return out
   260	    if check is None:
   261	        out["cause"] = "no check.json (queued, running, crashed or VOID)"
   262	        return out
   263	    want = dict(name=name, arm=arm, seed=seed, S=C.S, E=C.E, mode="continuation")
   264	    for key, value in want.items():
   265	        if check.get(key) != value:
   266	            raise InputError(f"{name}: check.json {key} = {check.get(key)!r}, expected {value!r}")
   267	    if check.get("passed") is not True or check.get("failures"):
   268	        out["cause"] = "CHECK failed (VOID until the cause is fixed and it reruns)"
   269	        return out
   270	    T, stop = check.get("T"), check.get("stop_reason")
   271	    if stop == "end" and T == C.E:
   272	        out.update(cls=COUNTED, T=T)
   273	    elif stop == "collapse" and isinstance(T, int) and not isinstance(T, bool) and C.S < T < C.E:
   274	        out.update(cls=STOPPED, T=T, cause=f"the collapse stop at T = {T:,} (no rows before E: filled)")
   275	    else:
   276	        raise InputError(f"{name}: a passing check.json with stop reason {stop!r} at T = {T!r}")
   277	    return out

   280	def gather(root):
   281	    """[(class dict, check)] in the criteria's queue order."""
   282	    root = Path(root)
   283	    lost = lost_of(root)
   284	    out = []
   285	    for arm, seed in QUEUE:
   286	        name = C.run_name(arm, seed)
   287	        check, csha = _read_json(root / "runs" / name / C.CHECK_JSON)
   288	        out.append((classify(name, arm, seed, check, csha, lost.get(name)), check))
   289	    return out

   310	def check_markers(m, c, check):
   311	    """markers.json against its class and check.json (raises InputError on a record that disagrees)."""
   312	    name = c["name"]
   313	    if not isinstance(m, dict):
   314	        raise InputError(f"{name}: no markers.json")
   315	    for key, value in (("schema", R.MARKERS_SCHEMA), ("kind", "continuation"), ("label", name), ("arm", c["arm"]),
   316	                       ("seed", c["seed"]), ("S", C.S), ("T", c["T"]),
   317	                       ("stop_reason", "end" if c["cls"] == COUNTED else "collapse")):
   318	        if m.get(key) != value:
   319	            raise InputError(f"{name}: markers.json {key} = {m.get(key)!r}, expected {value!r}")
   320	    probe = m.get("probe")
   321	    if not isinstance(probe, dict):
   322	        raise InputError(f"{name}: markers.json has no probe markers")
   323	    src = (check.get("sources") or {}).get("policy_probe.npz")
   324	    if not isinstance(src, list) or len(src) != 2 or probe.get("sha256") != src[1]:
   325	        raise InputError(f"{name}: markers.json was read from another policy_probe.npz than the one CHECK passed "
   326	                         f"({probe.get('sha256')!r} vs {src!r})")
   327	    steps = probe.get("steps")
   328	    want = [C.S + C.ROLLOUT * k for k in range(1, (c["T"] - C.S) // C.ROLLOUT)]
   329	    if steps != want or probe.get("n_rows") != len(want):
   330	        raise InputError(f"{name}: markers' probe rows are not S + 8,192 k in (S, T) ({probe.get('n_rows')!r} rows)")
   331	    for key in ("rollout_mu14_mean", "tail3_statue_settled_mean", "trailing8_mean"):
   332	        if not isinstance(probe.get(key), list) or len(probe[key]) != len(want):
   333	            raise InputError(f"{name}: markers' probe {key} is not one value per row")
   334	    blind = c["arm"] == INTERVENTION
   335	    form = "blind+raw" if blind else "plain"
   336	    for row in m.get("loop") or []:
   337	        if row.get("critic_form") != form:
   338	            raise InputError(f"{name}: markers' loop row {row.get('steps')} critic form {row.get('critic_form')!r} "
   339	                             f"is not {form!r} (the arm's reading)")
   340	        if blind:
   341	            for states, d in (row.get("critic_slope_obs6") or {}).items():
   342	                eff = (((d or {}).get("effective")) or {}).get("dV_dq3_per_deg")
   343	                if eff not in (None, 0.0):
   344	                    raise InputError(f"{name}: a critic_blind_tail3 run's effective critic slope on observation 6 "
   345	                                     f"is {eff} on the {states} states at {row.get('steps')} (must be 0.0)")
   346	    # the final pair's loop row (SEC-G, SEC-MARGIN) is the model at E: its recorded num_timesteps must be E
   347	    for row in loop_rows_at(m, C.E):
   348	        if row.get("num_timesteps") != C.E:
   349	            raise InputError(f"{name}: markers' loop row at E = {C.E:,} records num_timesteps "
   350	                             f"{row.get('num_timesteps')!r} (the final pair must be the model at E)")

   357	def mean_over(probe, key, steps):
   358	    """(Fraction mean, None) of probe[key] over the rows at the given steps, or (None, cause)."""
   359	    st = probe.get("steps") or []
   360	    vals = []
   361	    for s in steps:
   362	        idx = [i for i, x in enumerate(st) if x == s]
   363	        if len(idx) != 1:
   364	            return None, f"{len(idx)} probe rows at update_timesteps {s:,}"
   365	        v = probe[key][idx[0]]
   366	        if not _finite(v):
   367	            return None, f"{key} at {s:,} is {v!r}"
   368	        vals.append(Fraction(v))
   369	    return sum(vals, Fraction(0)) / len(vals), None

   382	def secondary_value(key, c, m, primary_cause):
   383	    """(Fraction, None) or (None, cause): one secondary's value for one continuation (criteria THE SECONDARIES).
   384	    MISSING for a LOST, STOPPED or UNREADABLE continuation, and when its marker is absent or not finite."""
   385	    if c["cls"] != COUNTED:
   386	        return None, c.get("cause") or c["cls"]
   387	    if primary_cause is not None:
   388	        return None, f"UNREADABLE ({primary_cause})"
   389	    if key in ("SEC-G", "SEC-MARGIN"):
   390	        rows = loop_rows_at(m, C.E)
   391	        if len(rows) != 1:
   392	            return None, f"{len(rows)} M-LOOP rows at E = {C.E:,} (the final pair)"
   393	        field = "G" if key == "SEC-G" else "b_star"
   394	        v, what = _get(rows[0], "OWN", "P", field), f"OWN.P.{field} at E"
   395	    elif key == "SEC-CMD":
   396	        per = _get(m, "ledger", "direct", "per_evaluation")
   397	        e = per.get(str(C.E)) if isinstance(per, dict) else None
   398	        if not isinstance(e, dict):
   399	            return None, f"the evaluation at E = {C.E:,} is absent from the ledger markers"
   400	        v, what = e.get("cmd_l"), "CMD_L at E"
   401	    else:
   402	        raise ValueError(key)
   403	    if not _finite(v):
   404	        return None, f"{what} is {v!r} (no finite value)"
   405	    return Fraction(v), None

   408	def ledger_report(m):
   409	    """Returns and falls in (S, T] from the ledger the markers name (report only; never fails the score).
   410	    (dict, provisional item or None)."""
   411	    lm = m.get("ledger") or {}
   412	    path, want = lm.get("file"), lm.get("sha256")
   413	    if not path:
   414	        return {"status": "no ledger named"}, None
   415	    try:
   416	        got = C.sha256_file(path)
   417	    except OSError as exc:
   418	        return {"status": f"unreadable ({exc})"}, None
   419	    if got != want:
   420	        return ({"status": "changed since the read (sha256 differs)"},
   421	                f"{m.get('label')}: the ledger {path} changed since the read ({got} != {want})")
   422	    per = {}
   423	    falls = []
   424	    try:
   425	        with open(path, newline="", encoding="utf-8") as fh:
   426	            rows = list(csv.DictReader(fh))
   427	    except (OSError, UnicodeDecodeError, csv.Error) as exc:
   428	        return {"status": f"unreadable ({exc.__class__.__name__}: {exc})"}, None
   429	    for r in rows:
   430	        try:
   431	            t = float(r.get("timesteps"))
   432	            ret = float(r.get("return"))
   433	        except (TypeError, ValueError):
   434	            continue
   435	        if not (math.isfinite(t) and t == int(t) and C.S < int(t) <= m["T"]):
   436	            continue
   437	        per.setdefault(int(t), []).append(ret)
   438	        if str(r.get("full_horizon")).strip() not in ("1", "1.0", "True", "true"):
   439	            falls.append({"timesteps": int(t), "episode": r.get("episode"), "length": r.get("length"),
   440	                          "cause": r.get("cause")})
   441	    rets = [x for v in per.values() for x in v]
   442	    return {"status": "read", "n_episodes": len(rets), "n_evaluations": len(per),
   443	            "mean_return_window": statistics.fmean(rets) if rets else None,
   444	            "mean_return_at_T": statistics.fmean(per[max(per)]) if per else None,
   445	            "mean_return_first": statistics.fmean(per[min(per)]) if per else None,
   446	            "mean_return_per_evaluation": {t: statistics.fmean(v) for t, v in sorted(per.items())},
   447	            "n_falls": len(falls), "falls": falls[:20]}, None

   468	def trajectory(probe, T):
   469	    """[(evaluation step t, Fraction or None)] for t in (S, T] on the 131,072 grid: the mean of rollout_mu14_mean over
   470	    the 8 probe rows ending at t (at t = T, which has no row, the 8 rows ending at T - 8,192: at E the primary's rows).
   471	    Report only."""
   472	    out = []
   473	    for t in range(C.S + C.EVAL_FREQ, T + 1, C.EVAL_FREQ):
   474	        end = t if t < T else T - C.ROLLOUT
   475	        v, _ = mean_over(probe, "rollout_mu14_mean", tuple(end - C.ROLLOUT * k for k in range(PRIMARY_K - 1, -1, -1)))
   476	        out.append((t, v))
   477	    return out

   480	def g_end_above_start(m):
   481	    """True / False: OWN G at the loop row at E above OWN G at the loop row at S (the start pair); None if either row
   482	    or value is absent (report only)."""
   483	    rs, re_ = loop_rows_at(m, C.S), loop_rows_at(m, C.E)
   484	    if len(rs) != 1 or len(re_) != 1:
   485	        return None
   486	    gs, ge = _get(rs[0], "OWN", "P", "G"), _get(re_[0], "OWN", "P", "G")
   487	    if not (_finite(gs) and _finite(ge)):
   488	        return None
   489	    return ge > gs
