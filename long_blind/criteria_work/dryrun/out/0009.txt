    84	def _probe(value, T, nan_at=None):
    85	    steps = [S + R8 * k for k in range(1, (T - S) // R8)]
    86	    mu = [float(B_S) + (value - B_S) * (i + 1) / len(steps) for i in range(len(steps))]
    87	    for i, s in enumerate(steps):                      # the last 8 rows carry the value exactly
    88	        if s in SB.PRIMARY_STEPS:
    89	            mu[i] = value
    90	    if nan_at is not None:
    91	        mu[steps.index(nan_at)] = float("nan")
    92	    ss = [0.05] * len(steps)
    93	    trail = [None] * 7 + [0.05] * (len(steps) - 7)
    94	    return {"file": "x", "sha256": FAKE_PROBE_SHA, "label": "x", "S": S, "T": T, "n_rows": len(steps), "steps": steps,
    95	            "rollout_mu14_mean": mu, "tail3_statue_settled_mean": ss, "trailing8_mean": trail,
    96	            "trailing8_first_above": {"0.10": None}, "trailing8_max": 0.05, "std14": [0.19] * len(steps),

   100	def _loop(arm, T, g=-0.5, b=1.0, final=True, num_ts_E=None):
   101	    """M-LOOP rows at S, the periodic pairs before T and (final) the final pair at T, as read_bs.saved_pairs gives."""
   102	    blind = arm == BLIND
   103	    rows = []
   104	    for s in (S, *[x for x in C.SAVED_STEPS if x < T], *([T] if final else [])):
   105	        end = s == T
   106	        G = g if end else (G_S if s == S else -0.6)
   107	        bstar = b if end else 1.4
   108	        crit = {k: {"raw": {"dV_dq3_per_deg": 0.01}, "effective": {"dV_dq3_per_deg": 0.0 if blind else 0.01}}
   109	                for k in ("REF", "OWN", "ONPOLICY")}
   110	        rows.append({"steps": s, "num_timesteps": num_ts_E if (end and num_ts_E is not None) else s,
   111	                     "critic_form": "blind+raw" if blind else "plain",
   112	                     "OWN": {"P": {"G": G, "c0": 1.0 - G - bstar, "b_star": bstar}, "L": {"G": -0.8},
   113	                             "own14_median": 0.0, "b_star_L": 1.8, "n_terminated": 0},
   114	                     "REF": {"P": {"G": -0.7, "c0": -0.1}}, "critic_slope_obs6": crit})
   115	    return rows

   145	class Screen:
   146	    """A synthetic <root> with SCREEN.json and, per entry, any of: check.json, run_meta.json, markers.json, ledger."""
   147	
   148	    def __init__(self, root):
   149	        self.root = Path(root)
   150	        _write(self.root / "SCREEN.json", {"schema": C.SCREEN_SCHEMA, "intervention": BLIND,

   153	    def run(self, arm, seed, value, *, stop="end", T=E, check=True, read=True, passed=True, nan_at=None,
   154	            meta_kw=None, g=-0.5, cmd=0.5, cmd_max=0.9, b=1.0, final=True, drop_E=False, num_ts_E=None):
   155	        name = C.run_name(arm, seed)
   156	        rd = self.root / "runs" / name
   157	        meta = _write(rd / C.RUN_META, _meta(arm, seed, self.root, **(meta_kw or {})))
   158	        led = _ledger(rd / "01_stance" / "eval_episodes.csv", T, falls=1 if arm == BLIND else 0)
   159	        if check:
   160	            _write(rd / C.CHECK_JSON, {
   161	                "schema": R.CHECK_SCHEMA, "tool_sha256": _sha(TOOLS / "read_bs.py"), "name": name, "mode":
   162	                "continuation", "passed": passed, "failures": [] if passed else ["x"], "arm": arm, "seed": seed,
   163	                "S": S, "E": E, "T": T, "end": E, "stop_reason": stop,
   164	                "sources": {"run_meta.json": [str(meta), _sha(meta)], "policy_probe.npz": ["p", FAKE_PROBE_SHA]}})
   165	        if read:
   166	            direct = _direct(T, cmd, cmd_max, drop_E)
   167	            _write(self.root / "reads" / name / "markers.json", {
   168	                "schema": R.MARKERS_SCHEMA, "tool_sha256": _sha(TOOLS / "read_bs.py"), "label": name,
   169	                "kind": "continuation", "arm": arm, "seed": seed, "S": S, "T": T,
   170	                "stop_reason": stop, "probe": _probe(value, T, nan_at),
   171	                "ledger": {"file": str(led), "sha256": _sha(led), "direct": direct,
   172	                           "m5": {"any_flagged": [], "tail_flagged": [], "any_alarms": [], "tail_alarms": [],
   173	                                  "n_c6_counting_failures": 0}},
   174	                "loop": _loop(arm, T, g, b, final, num_ts_E)})
   175	        return name

   177	    def all(self, control, blind, per_seed=None, **kw):
   178	        """per_seed: {(arm, seed): {run() keyword: value}} for the secondaries and the odd cases."""

   195	def _brute_p(higher, other):
   196	    """An independent float implementation of the test (no fill): the share of relabelings whose `higher` sum is at
   197	    least the observed."""
   198	    vals = list(higher) + list(other)
   199	    obs = sum(higher)
   200	    subs = list(itertools.combinations(range(len(vals)), len(higher)))
   201	    return sum(1 for s in subs if sum(vals[i] for i in s) >= obs - 1e-12) / len(subs)
