  1022	def loop_schedule(T, every=None):
  1023	    """The steps the M-LOOP reads in [S, T]: S, the saved steps < T whose (steps - S) is a multiple of every
  1024	    (bs_common.LOOP_READ_EVERY), and T (the final pair) (pure)."""
  1025	    every = int(every or C.LOOP_READ_EVERY)
  1026	    if every % C.SAVE_FREQ:
  1027	        raise InputError(f"LOOP_READ_EVERY {every} is not a multiple of the save grid {C.SAVE_FREQ}")
  1028	    return (C.S, *[x for x in C.SAVED_STEPS if x < T and (x - C.S) % every == 0], T)

  1031	def saved_pairs(run_dir, T, every=None):
  1032	    """[(steps, zip, pkl)] in [S, T] on loop_schedule: the start copy, the periodic pairs it names, and the final
  1033	    pair at T."""

  1041	    if z.is_file() and p.is_file():

  1088	def loop_line(row):
  1089	    def g(*ks):
  1090	        d = row
  1091	        for k in ks:
  1092	            if not isinstance(d, dict) or k not in d:
  1093	                return None
  1094	            d = d[k]
  1095	        return d
  1096	
  1097	    def f(x, fmt="+.3f"):
  1098	        return "  n/a " if x is None else format(x, fmt)
  1099	    return (f"OWN G {f(g('OWN', 'P', 'G'))} c0 {f(g('OWN', 'P', 'c0'))} own14 {f(g('OWN', 'own14_median'))} "
  1100	            f"G_L {f(g('OWN', 'L', 'G'))} b* {f(g('OWN', 'P', 'b_star'))} | REF G {f(g('REF', 'P', 'G'))} c0 "
  1101	            f"{f(g('REF', 'P', 'c0'))} | dV/dq3 on-policy eff "
  1102	            f"{f(g('critic_slope_obs6', 'ONPOLICY', 'effective', 'dV_dq3_per_deg'), '+.5f')} "
  1103	            f"raw {f(g('critic_slope_obs6', 'ONPOLICY', 'raw', 'dV_dq3_per_deg'), '+.5f')} own eff "
  1104	            f"{f(g('critic_slope_obs6', 'OWN', 'effective', 'dV_dq3_per_deg'), '+.5f')}")
