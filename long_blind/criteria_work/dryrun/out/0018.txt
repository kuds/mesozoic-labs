   909	def _steps(xs, n=6):
   910	    xs = list(xs)
   911	    s = ", ".join(f"{x:,}" if isinstance(x, int) else str(x) for x in xs[:n])
   912	    return s + (f", ... ({len(xs)} in all)" if len(xs) > n else "")

   981	def cmd_classify(a) -> int:
   982	    out_root = require_under(Path(a.out_root))
   983	    out_root.mkdir(parents=True, exist_ok=True)
   984	    problems = check_pins()
   985	    if problems:
   986	        print("REFUSED: pins:\n  " + "\n  ".join(problems), file=sys.stderr)
   987	        return 2
   988	    C, LG, CP, SC = lane_b()
   989	    lock = threading.Lock()
   990	    biglog = require_under(out_root / "measure_lb.log")
   991	
   992	    def log(msg):
   993	        line = f"{utcnow()} {msg}"
   994	        with lock:
   995	            print(line, flush=True)
   996	            with open(biglog, "a") as fh:
   997	                fh.write(line + "\n")
   998	
   999	    log(f"{VERSION}; runs {a.runs}; slots {a.slots}; prune {a.prune_csv}; simulate {not a.no_simulate}; "
  1000	        f"free {free_bytes(out_root):,} bytes; {MPL_DIR_OUTSIDE} exists {os.path.isdir(MPL_DIR_OUTSIDE)}")
  1001	    preps, rc = [], 0
  1002	    for rd in a.runs:
  1003	        try:
  1004	            t0 = time.monotonic()
  1005	            prep = prepare(C, LG, CP, Path(rd), a.suffix, out_root, a.start)
  1006	            prep["prep_s"] = round(time.monotonic() - t0, 2)
  1007	            prep["m5cli"] = run_m5_cli(prep)
  1008	            log(f"{prep['name']}: E {prep['E']:,} ({prep['ending']['ending']}), labels {prep['labels']}, "
  1009	                f"M5 CLI {prep['m5cli']['wall_s']} s")
  1010	            preps.append(prep)
  1011	        except Refusal as exc:
  1012	            log(f"{rd}: REFUSED: {exc.message}")
  1013	            rc = max(rc, exc.code)
  1014	    jobs = [j for p in preps for j in jobs_for(p, not a.no_simulate, a.prune_csv)]
  1015	    todo = [j for j in jobs if "outcome" not in j]           # per run: the certificate first (longest), then traces
  1016	    slots = Slots(a.slots, log)
  1017	    done = [j for j in jobs if "outcome" in j]
  1018	    t0 = time.monotonic()
  1019	    if todo:
  1020	        log(f"launching {len(todo)} measurement(s) on {a.slots} slot(s)")
  1021	        with ThreadPoolExecutor(max_workers=a.slots) as ex:
  1022	            done += list(ex.map(lambda j: run_job(j, slots, log), todo))
  1023	    log(f"measurements done in {time.monotonic() - t0:.0f} s wall; {MPL_DIR_OUTSIDE} exists "
  1024	        f"{os.path.isdir(MPL_DIR_OUTSIDE)}")
  1025	    parent_ledger = Path(a.parent_ledger) if a.parent_ledger else None
  1026	    summary = []
  1027	    for p in preps:
  1028	        jr = [j for j in done if j["run"] == p["name"]]
  1029	        for j in jr:
  1030	            j.pop("logs", None)
  1031	        if any(j["outcome"] in ("CRASHED TWICE",) for j in jr):
  1032	            rc = max(rc, 3)
  1033	        t1 = time.monotonic()
  1034	        res = classify_one(p, jr, p["m5cli"], parent_ledger, a.prune_csv)
  1035	        res["costs"] = dict(prepare_s=p["prep_s"], m5_cli_s=p["m5cli"]["wall_s"],
  1036	                            classify_s=round(time.monotonic() - t1, 2),
  1037	                            jobs={f"{j['label']} {j['kind']}": {k: j["attempts"][-1].get(k) for k in
  1038	                                                                 ("wall_s", "cpu_s", "cpu_user_s", "cpu_sys_s",
  1039	                                                                  "maxrss_kb")} if j["attempts"] else j["outcome"]
  1040	                                  for j in jr},
  1041	                            m1_cpu_s=round(sum(j["attempts"][-1].get("cpu_s") or 0 for j in jr
  1042	                                               if j["kind"] == "m1" and j["attempts"]), 1),
  1043	                            m3_cpu_s=round(sum(j["attempts"][-1].get("cpu_s") or 0 for j in jr
  1044	                                               if j["kind"] == "m3" and j["attempts"]), 1))
  1045	        res["free_bytes_after"] = free_bytes(out_root)
  1046	        res["mpl_dir_outside_exists"] = os.path.isdir(MPL_DIR_OUTSIDE)
  1047	        sha = write_json(p["out"] / "measure_lb.json", res)
  1048	        text = render(res)
  1049	        write_text(p["out"] / "report.txt", text + "\n")
  1050	        if res.get("classes") is not None:
  1051	            write_json(p["out"] / "classes.json", res["classes"])
  1052	            write_json(p["out"] / "habit.json", res["habit"])
  1053	        print(text, flush=True)
  1054	        log(f"{p['name']}: {res['status']}; measure_lb.json {sha[:16]}")
  1055	        if res["status"] != "CLASSIFIED":
  1056	            rc = max(rc, 2)
  1057	        c = res.get("classes") or {}
  1058	        summary.append(dict(run=p["name"], status=res["status"], cls=c.get("cls"), kind=c.get("kind"),
  1059	                            gate=c.get("gate"), outcome=c.get("outcome"),
  1060	                            tail_alarms=len(c.get("tail_alarms") or []),
  1061	                            first_tail_alarm=(c.get("tail_alarms") or [None])[0],
  1062	                            any_alarms=len(c.get("any_alarms") or []),
  1063	                            habit_sites=(res.get("habit") or {}).get("habit_sites")))
  1064	    write_json(out_root / "summary.json", dict(utc=utcnow(), version=VERSION, runs=summary, rc=rc))
  1065	    return rc

  1068	def cmd_plan(a) -> int:
  1069	    """Inputs, ending, M5 (in process only) and the jobs C7 names, with the disk they need; nothing simulated and
  1070	    nothing written.  It reads the ledger, so the outcome-blind guard applies."""
  1071	    problems = check_pins()
  1072	    if problems:
  1073	        print("REFUSED: pins:\n  " + "\n  ".join(problems), file=sys.stderr)
  1074	        return 2
  1075	    C, LG, CP, SC = lane_b()
  1076	    rc = 0
  1077	    for rd in a.runs:
  1078	        try:
  1079	            p = prepare(C, LG, CP, Path(rd), a.suffix, require_under(Path(a.out_root)), a.start)
  1080	        except Refusal as exc:
  1081	            print(f"{rd}: REFUSED: {exc.message}")
  1082	            rc = max(rc, exc.code)
  1083	            continue
  1084	        jobs = jobs_for(p, True)
  1085	        new = [j for j in jobs if "outcome" not in j]
  1086	        need_b = sum(M3_OUT_BYTES if j["kind"] == "m3" else M1_OUT_BYTES for j in new)
  1087	        print(json.dumps(dict(run=p["name"], E=p["E"], ending=p["ending"]["ending"], S=p["S"], labels=p["labels"],
  1088	                              resolution={str(k): v for k, v in p["resolution"].items()},
  1089	                              c6_passed=p["c6"]["passed"], jobs=[f"{j['label']} {j['kind']}: "
  1090	                                                                 f"{j.get('outcome', 'TO MEASURE')}" for j in jobs],
  1091	                              output_bytes_before_prune=need_b, free_bytes=free_bytes(p["out"].parent),
  1092	                              disk_floor=DISK_FLOOR), indent=1))
  1093	    return rc
