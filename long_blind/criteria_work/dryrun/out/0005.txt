     1	"""Bias screen 2 (BS2): the scorer.  Report only: reads files, never trains, never simulates.
     2	
     3	    PY -I -B score_bs.py [--root <BS2>] [--out <root>/score]
     4	
     5	It applies <BS2>/criteria.txt (BS2 criteria v1) exactly (THE PRIMARY OUTCOME, THE TEST, THE INFORMATIVENESS CONDITION,
     6	THE READING, THE SECONDARIES, REPORT ONLY, RUN HANDLING, WHEN THE READING IS MADE); the constants below are the text's
     7	and are pinned with it (criteria.txt names this file's sha256 in PINS; <BS2>/APPROVED pins it as
     8	tool_sha256["score_bs.py"]).  Derived from BS's score_bs.py (score_bs 1.1, sha256 869898c5...): the window, B_S, the
     9	gate (0.20), the queue's seeds and the line texts are BS2's; the three pre-registered secondaries, the report-only
    10	trajectory and the count of G(E) > G(S) are new; the code copy is measured at bs_common.CODE (BS's copy, reused).
    11	
    12	What it reads: <root>/SCREEN.json (it must be the criteria's queue); <root>/lost.json; for each queue entry
    13	<root>/runs/<name>/check.json (read_bs.py --check).  Only once every entry is decided (COUNTED, STOPPED or LOST) and
    14	every COUNTED or STOPPED one has its full read: <root>/reads/<name>/markers.json (read_bs.py's markers), the run's
    15	run_meta.json (the per-rollout manipulation evidence summary and the pins it ran under) and the ledger the markers name
    16	(returns and falls, report only).  For PROVISIONAL: <root>/APPROVED, <root>/criteria.txt, <root>/SCREEN.json, the five
    17	BS2 tools and this file (in this file's folder) against APPROVED, M5's pins, and the code copy bs_common.CODE measured
    18	by M1's own r10m_common.check_copy (loaded only at its pin).
    19	What it writes: <out>/score.json and <out>/score.txt (default <root>/score; refused outside <root>).
    20	
    21	Classes: LOST (named in lost.json); COUNTED (check.json passed, stop reason "end" at E); STOPPED (check.json passed,
    22	stop reason "collapse" at T < E); UNDECIDED otherwise (running, queued, crashed or VOID).  While any entry is UNDECIDED,
    23	or a COUNTED / STOPPED one has no markers.json yet, the screen is "incomplete": no markers file and no run_meta.json is
    24	opened and no outcome is printed (outcome-blind).
    25	
    26	THE PRIMARY (one number per COUNTED continuation): the mean of the probe's rollout_mu14_mean (policy_probe.npz
    27	rollout_mu_mean[:, 14], the actor's raw Gaussian mean for tail_3_pitch_act averaged over the rollout's own 8,192
    28	states) over the 8 probe rows with update_timesteps E - 65,536 ... E - 8,192 (3,473,408 ... 3,530,752), in exact
    29	rational arithmetic.  MISSING (filled): a LOST or STOPPED continuation, or a COUNTED one whose 8 rows are not each
    30	present once and finite ("primary unreadable", UNREADABLE).
    31	THE TEST: exact one-sided permutation test of control against critic_blind_tail3 on the difference of arm means, over
    32	all C(8, 4) = 70 relabelings; p = (relabelings whose control-arm sum is >= the observed) / 70, exact rationals (a tie
    33	counts against the reading).
    34	THE FILL: a missing control value counts as -infinity and a missing critic_blind_tail3 value as +infinity (the supremum
    35	of p over every value the missing ones could have had: it never helps REDUCED).
    36	THE READING, the first line that applies: 1 NOT INFORMATIVE (the median control primary rose by less than 0.20 from
    37	B_S, or no control primary was read; the median rule: the midpoint of the two middle values for 4, the middle one for
    38	3, the midpoint for 2, the value for 1, exact); 2 REDUCED (filled p <= 1/20); 3 NO REDUCTION DETECTED, with "(the
    39	as-counted p, report only, is <= 0.05; the filled p decides)" when the as-counted p is <= 1/20 and the filled one is
    40	not.
    41	THE SECONDARIES (tested, never change the line): SEC-G (OWN G at the final pair, E; blind HIGHER), SEC-CMD (CMD_L at the
    42	evaluation at E; blind LOWER), SEC-MARGIN (OWN b*_P = 1 - G - c0 at the final pair, E; blind HIGHER).  Each: the same
    43	exact test on its own values with the hypothesis arm's sum as the statistic, its own fill (a MISSING value counts as the
    44	infinity that works against the secondary's direction), SUPPORTED when the filled p <= 0.05 (unadjusted), with
    45	"(survives Bonferroni over the 3 secondaries)" when the filled p <= 0.05 / 3 (only 1/70 can), and, under line 1,
    46	"(the control arm did not climb on the primary)".  A secondary is MISSING for a LOST, STOPPED or UNREADABLE
    47	continuation and when its marker is absent or not finite (cause printed); a COUNTED continuation whose markers lack the
    48	loop row at E (the final pair not read: a tool or file defect) is also a PROVISIONAL item (score_bs 2.1, finding D7).
    49	Each arm's sample SD of the primaries read is printed beside its mean and median (report only: it places the power).
    50	
    51	Exit codes (as score_ecs.py): 0 final; 3 provisional or incomplete; 2 input error.

   159	def subset_sums(values, k):
   160	    """{subset (tuple of indices): sum} over all k-subsets; values are (infinite part, finite Fraction) pairs."""
   161	    out = {}
   162	    for sub in itertools.combinations(range(len(values)), k):
   163	        inf = sum(values[i][0] for i in sub)
   164	        fin = sum((values[i][1] for i in sub), Fraction(0))
   165	        out[sub] = (inf, fin)
   166	    return out

   169	def perm_test(higher, other):
   170	    """The exact one-sided permutation test that the arm `higher` is higher (for the primary: control higher = the
   171	    blind arm lower).  higher, other: lists of (infinite part, Fraction) pairs.  Returns (p, count, total): p =
   172	    #{relabelings whose `higher`-arm sum is >= the observed} / total (a tie counts against the reading), as an exact
   173	    Fraction; the sum of one arm is equivalent to the difference of arm means for any arm sizes.  None when an arm is
   174	    empty."""
   175	    if not higher or not other:
   176	        return None
   177	    values = list(higher) + list(other)
   178	    k = len(higher)
   179	    obs = (sum(v[0] for v in higher), sum((v[1] for v in higher), Fraction(0)))
   180	    sums = subset_sums(values, k)
   181	    count = sum(1 for s in sums.values() if s >= obs)
   182	    return Fraction(count, len(sums)), count, len(sums)

   185	def sign_flip_test(diffs):
   186	    """Exact one-sided sign-flip test of paired differences (control - blind; report only).  (p, count, total)."""
   187	    if not diffs:
   188	        return None
   189	    obs = sum(diffs, Fraction(0))
   190	    total = 0
   191	    count = 0
   192	    for signs in itertools.product((1, -1), repeat=len(diffs)):
   193	        total += 1
   194	        if sum((s * d for s, d in zip(signs, diffs, strict=True)), Fraction(0)) >= obs:
   195	            count += 1
   196	    return Fraction(count, total), count, total

   211	def _read_json(path):
   212	    """(object, sha256) or (None, None) when the file does not exist."""
   213	    try:
   214	        raw = Path(path).read_bytes()
   215	    except FileNotFoundError:
   216	        return None, None
   217	    except OSError as exc:
   218	        raise InputError(f"cannot read {path}: {exc}") from exc
   219	    try:
   220	        d = json.loads(raw.decode("utf-8"))
   221	    except (UnicodeDecodeError, ValueError) as exc:
   222	        raise InputError(f"{path} is not JSON: {exc}") from exc
   223	    if not isinstance(d, dict):
   224	        raise InputError(f"{path} is not a JSON object")
   225	    return d, hashlib.sha256(raw).hexdigest()

   228	def screen_of(root):
   229	    """SCREEN.json, valid (bs_common.screen_problems) and equal to the criteria's queue."""
   230	    d, problems = C.load_screen(str(Path(root) / "SCREEN.json"))
   231	    if problems:
   232	        raise InputError("SCREEN.json: " + "; ".join(problems))
   233	    if d.get("intervention") != INTERVENTION:
   234	        raise InputError(f"SCREEN.json intervention {d.get('intervention')!r} is not the criteria's {INTERVENTION}")
   235	    if C.queue_of(d) != QUEUE:
   236	        raise InputError(f"SCREEN.json queue {C.queue_of(d)} is not the criteria's {QUEUE}")
   237	    return d

   240	def lost_of(root):
   241	    try:
   242	        lost = R.load_lost(Path(root) / "lost.json")
   243	    except R.InputError as exc:
   244	        raise InputError(str(exc)) from exc
   245	    names = {C.run_name(a, s) for a, s in QUEUE}
   246	    unknown = sorted(set(lost) - names)
   247	    if unknown:
   248	        raise InputError(f"lost.json names continuations outside the queue: {unknown}")
   249	    return lost
