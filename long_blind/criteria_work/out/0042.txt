def loop_schedule(T, every=None):
    """The steps the M-LOOP reads in [S, T]: S, the saved steps < T whose (steps - S) is a multiple of every
    (bs_common.LOOP_READ_EVERY), and T (the final pair) (pure)."""
    every = int(every or C.LOOP_READ_EVERY)
    if every % C.SAVE_FREQ:
        raise InputError(f"LOOP_READ_EVERY {every} is not a multiple of the save grid {C.SAVE_FREQ}")
    return (C.S, *[x for x in C.SAVED_STEPS if x < T and (x - C.S) % every == 0], T)


def saved_pairs(run_dir, T, every=None):
    """[(steps, zip, pkl)] in [S, T] on loop_schedule: the start copy, the periodic pairs it names, and the final
    pair at T."""
    models = Path(run_dir) / MODELS_REL
    out = []
    for s in loop_schedule(T, every)[:-1]:
        z, p = (models / n for n in C.periodic_names(s))
        if z.is_file() and p.is_file():
            out.append((s, z, p))
    z, p = models / C.FINAL_ZIP, models / C.FINAL_PKL
    if z.is_file() and p.is_file():
        out.append((T, z, p))
    return out
